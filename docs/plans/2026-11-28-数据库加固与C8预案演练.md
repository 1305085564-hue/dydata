## 可复跑的双身份演练

在仓库根目录执行下面整段命令。两个独立内存库只使用合成数据。身份、业务表和可见范围使用最小夹具；员工通过本人行策略写入，两份正式迁移、审批、重开、审计和约束均执行原文件。

```bash
node --input-type=module <<'JS'
import assert from 'node:assert/strict';
import {readFileSync} from 'node:fs';
import {PGlite} from '@electric-sql/pglite';

const migration = name => readFileSync('supabase/migrations/' + name, 'utf8');
const base = migration('20261127100000_late_submit_post_review.sql');
const harden = migration('20261128100000_harden_settlement_and_post_review.sql');
const ids = Array.from({length: 14}, (_, n) =>
  "'00000000-0000-4000-8000-" + String(n + 1).padStart(12, '0') + "'");
const [member, admin, other, account, video, report, appeal, otherVideo, otherReport,
  extra, newAppeal, secondAccount, oldAppeal, spare] = ids;
for (const type of ['text', 'uuid']) {
  const db = new PGlite();
  const exec = sql => db.exec(sql);
  const rows = async sql => (await db.query(sql)).rows;
  const deny = async (sql, code) => assert.rejects(() => exec(sql), e => e.code === code);
  const role = async (name, user) => exec(`reset role; set role ${name}; set fixture.uid = ${user};`);
  const insertAppeal = (id, mode, v, r, acc = account) =>
    `insert into fulfillment_appeals(id,user_id,account_id,review_mode,video_id,daily_report_id)
     values(${id},${member},${acc},'${mode}',${v},${r})`;
  await exec(`
    create role anon; create role authenticated; create role service_role bypassrls;
    create schema auth; grant usage on schema auth to authenticated,service_role;
    create function auth.uid() returns ${type} language sql stable as
      $$select nullif(current_setting('fixture.uid',true),'')::${type}$$;
    create function auth.role() returns text language sql stable as $$select current_setting('role')$$;
    create table profiles(id uuid primary key,role text,company_role text,membership_status text);
    create function active_visible_user_ids(uuid) returns table(user_id uuid) language sql as
      $$select id from profiles where membership_status='active'$$;
    create table videos(id uuid primary key,user_id uuid,account_id uuid,title text);
    create table daily_reports(id uuid primary key,user_id uuid,account_id uuid,video_id uuid,play_count int,is_void boolean default false);
    create table fulfillment_appeals(id uuid primary key,user_id uuid,account_id uuid,record_date date,
      status text default 'pending',handler_id uuid,handled_at timestamptz,decision_reason text);
    create table audit_logs(user_id uuid,action text,target text,detail text);
    insert into profiles values (${member},'member','member','active'),(${admin},'admin','admin','active');
    insert into videos values (${video},${member},${account},'初稿'),(${otherVideo},${other},${account},'旁证');
    insert into daily_reports(id,user_id,account_id,video_id) values
      (${report},${member},${account},${video}),(${otherReport},${other},${account},${otherVideo}),
      (${extra},${member},${account},${video});
  `);
  await exec(base);
  await exec(insertAppeal(appeal, 'post_submit', otherVideo, report));
  await assert.rejects(() => db.transaction(tx => tx.exec(harden)), e => e.code === '23503');
  await exec('delete from fulfillment_appeals');
  await exec(harden);
  for (const table of ['videos','daily_reports']) {
    await exec(`alter table ${table} enable row level security;
      create policy edit_own on ${table} for all to authenticated using (user_id=(auth.uid())::uuid)
        with check (user_id=(auth.uid())::uuid);
      grant select,insert,update on ${table} to authenticated,service_role;`);
  }
  await role('authenticated', member);
  for (const [table,key] of [['videos',video],['daily_reports',report]]) {
    for (const [field,value] of Object.entries({settlement_excluded:'true', settlement_excluded_reason:"'人工'",
      settlement_excluded_at:'now()', settlement_excluded_by:member})) {
      await deny(`insert into ${table}(id,user_id,account_id,${field}) values(${spare},${member},${account},${value})`, '42501');
      await deny(`update ${table} set ${field}=${value} where id=${key}`, '42501');
    }
    assert.equal((await rows(`insert into ${table}(id,user_id,account_id) values(${spare},${member},${account}) returning settlement_excluded`))[0].settlement_excluded,false);
  }
  assert.equal((await rows(`update videos set title='定稿' where id=${video} returning title`))[0].title,'定稿');
  assert.equal((await rows(`update daily_reports set play_count=88 where id=${report} returning play_count`))[0].play_count,88);
  await exec('reset role');
  for (const [v,r,acc] of [[otherVideo,report,account],[video,otherReport,account],[video,report,secondAccount],
      [spare,report,account]]) {
    await deny(insertAppeal(newAppeal,'post_submit',v,r,acc),'23503');
  }
  await deny(insertAppeal(newAppeal,'post_submit',video,'null'),'23514');
  await exec(insertAppeal(appeal,'post_submit',video,report));
  await deny(`update daily_reports set video_id=${otherVideo} where id=${report}`,'23503');
  await deny(`update videos set account_id=${secondAccount} where id=${video}`,'23503');
  await exec(insertAppeal(oldAppeal,'pre_submit','null','null'));
  const handle = (id, decision) => `select handle_fulfillment_appeal(${id},'${decision}',null,'核查原因')`;
  const reopen = id => `select reopen_fulfillment_appeal_atomically(${id},null)`;
  const check = async (state,excluded) => {
    const [a] = await rows(`select status from fulfillment_appeals where id=${appeal}`);
    assert.equal(a.status,state);
    for (const [table,id] of [['videos',video],['daily_reports',report]]) {
      const [r] = await rows(`select settlement_excluded as flag, settlement_excluded_reason as reason,
        settlement_excluded_at is not null as stamped, settlement_excluded_by as actor from ${table} where id=${id}`);
      assert.deepEqual(r,{flag:excluded,reason:excluded?'核查原因':null,stamped:excluded,actor:excluded?admin.slice(1,-1):null});
    }
    assert.equal((await rows(`select settlement_excluded from daily_reports where id=${extra}`))[0].settlement_excluded,false);
  };
  await role('authenticated',member);
  await deny(handle(appeal,'reject'),'42501');
  await role('authenticated',admin);
  await exec(handle(appeal,'reject')); await exec('reset role'); await check('rejected',true);
  await role('authenticated',admin);
  await exec(reopen(appeal)); await exec('reset role'); await check('pending',false);
  await role('authenticated',admin);
  await exec(handle(appeal,'approve')); await exec('reset role'); await check('approved',false);
  await role('authenticated',admin);
  await exec(reopen(appeal)); await exec(handle(oldAppeal,'approve')); await exec(reopen(oldAppeal));
  await exec('reset role'); await check('pending',false);
  assert.equal((await rows('select count(*)::int as n from audit_logs'))[0].n,6);
  await role('service_role',admin);
  assert.equal((await rows(`update videos set settlement_excluded_reason='维护' where id=${spare}
    returning settlement_excluded_reason`))[0].settlement_excluded_reason,'维护');
  console.log(type, 'PASS · 双表16项写保护、普通提交编辑、存量及新增错绑、父行变更、审批/重开、旧单、审计、服务角色');
  await db.close();
}
```
