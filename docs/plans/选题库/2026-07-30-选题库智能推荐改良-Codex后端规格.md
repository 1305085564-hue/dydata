### 2.2 新增 `calcTopicScore()` 评分函数

两个 Tab（trending / high_potential）共用同一套公式，仅时间分方向相反。

```typescript
type ScoreMode = "trending" | "high_potential";

function calcFlowScore(avgPlayCount: number | null): number {
  const play = avgPlayCount ?? 0;
  if (play >= 200000) return 0.6;
  if (play >= 100000) return 0.5;
  if (play >= 50000)  return 0.4;
  if (play >= 30000)  return 0.3;
  return 0.1;
}

function calcTimeScore(daysSinceLastWork: number, mode: ScoreMode): number {
  if (mode === "trending") {
    // 奖励新鲜：越近分越高
    if (daysSinceLastWork <= 3)  return 0.4;
    if (daysSinceLastWork <= 7)  return 0.3;
    return 0.2; // 8~30 天
  } else {
    // 奖励沉睡：越久分越高
    if (daysSinceLastWork > 60)  return 0.4;
    if (daysSinceLastWork > 45)  return 0.3;
    return 0.2; // 30~45 天
  }
}

function calcTopicScore(
  avgPlayCount: number | null,
  daysSinceLastWork: number,
  mode: ScoreMode,
): number {
  return calcFlowScore(avgPlayCount) + calcTimeScore(daysSinceLastWork, mode);
}
```

### 2.3 新增 `loadScoredTopicPool()` 处理 trending / high_potential

```typescript
async function loadScoredTopicPool(
  supabase: TopicSupabase,
  userId: string,
  scope: DataAccessScope,
  options: TopicPoolQueryOptions,
  mode: ScoreMode,
): Promise<ApiResult<unknown>> {
  const now = Date.now();
  const MS_PER_DAY = 24 * 60 * 60 * 1000;

  // 1. 拉取所有带关联作品的子题（不分页，后端评分排序后再截取）
  const { data: subTopics, error: subTopicsError } = await supabase
    .from("sub_topics")
    .select("id, title, hook, emotion_tag, topic_id, group_id, created_by, created_at, topics(id, name, sort_order), topic_groups(id, name, sort_order), sub_topic_claims(id, user_id, status, claimed_at)");
  if (subTopicsError) return { ok: false, status: 500, message: subTopicsError.message };

  const allSubTopics = (subTopics ?? []) as Array<Record<string, unknown>>;
  const subTopicIds = allSubTopics.map((item) => String(item.id));
  if (!subTopicIds.length) {
    return { ok: true, value: { items: [], pagination: { page: 1, pageSize: options.pageSize, totalItems: 0 } } };
  }

  // 2. 拉取关联作品（取最新上传时间 + 播放量汇总）
  let worksQuery = supabase
    .from("videos")
    .select("topic_id, user_id, uploaded_at, video_metrics_snapshots(play_count)")
    .eq("lifecycle_state", "active")
    .in("topic_id", subTopicIds);
  if (scope.kind !== "all") worksQuery = worksQuery.in("user_id", scope.visibleUserIds);
  const { data: works, error: worksError } = await worksQuery;
  if (worksError) return { ok: false, status: 500, message: worksError.message };

  // 3. 按子题聚合：最新发布时间 + 平均播放
  const worksBySubTopic = new Map<string, { latestUploadedAt: string; playCounts: number[] }>();
  for (const work of (works ?? []) as Array<Record<string, unknown>>) {
    const topicId = String(work.topic_id ?? "");
    if (!topicId) continue;
    const uploadedAt = typeof work.uploaded_at === "string" ? work.uploaded_at : "";
    const snapshots = Array.isArray(work.video_metrics_snapshots) ? work.video_metrics_snapshots as Array<{ play_count?: number | null }> : [];
    const playCount = snapshots.reduce((max, s) => Math.max(max, Number(s.play_count ?? 0)), 0);

    const current = worksBySubTopic.get(topicId);
    if (!current) {
      worksBySubTopic.set(topicId, { latestUploadedAt: uploadedAt, playCounts: [playCount] });
    } else {
      if (uploadedAt > current.latestUploadedAt) current.latestUploadedAt = uploadedAt;
      current.playCounts.push(playCount);
    }
  }

  // 4. 根据 mode 过滤 + 评分
  const TRENDING_CUTOFF_DAYS = 30;
  const scored: Array<{ item: Record<string, unknown>; score: number; daysSince: number; avgPlay: number | null }> = [];

  for (const item of allSubTopics) {
    const id = String(item.id);
    const agg = worksBySubTopic.get(id);

    if (!agg) continue; // never_worked 在另一个函数处理，这里跳过无作品的

    const qualified = agg.playCounts.filter((p) => p >= 1000);
    const avgPlay = qualified.length ? Math.round(qualified.reduce((s, p) => s + p, 0) / qualified.length) : null;
    const latestDate = Date.parse(agg.latestUploadedAt) || 0;
    const daysSince = latestDate ? Math.floor((now - latestDate) / MS_PER_DAY) : 999;

    // trending：只要 ≤30 天的
    if (mode === "trending" && daysSince > TRENDING_CUTOFF_DAYS) continue;
    // high_potential：只要 >30 天的
    if (mode === "high_potential" && daysSince <= TRENDING_CUTOFF_DAYS) continue;

    // 没有合格作品的低质子题不进推荐（流量分 0.1 的也放进来，用户可以看到刚起步的）
    const score = calcTopicScore(avgPlay, daysSince, mode);
    scored.push({ item, score, daysSince, avgPlay });
  }

  // 5. 按综合分排序
  scored.sort((a, b) => b.score - a.score || (b.avgPlay ?? 0) - (a.avgPlay ?? 0));

  // 6. 分页截取
  const totalItems = scored.length;
  const from = (options.page - 1) * options.pageSize;
  const pageItems = scored.slice(from, from + options.pageSize);

  // 7. 加载这批子题的 summary
  const pageIds = pageItems.map((x) => String(x.item.id));
  let summaries: Map<string, TopicWorkSummary>;
  try {
    summaries = await loadTopicSummaries(supabase, pageIds, scope);
  } catch (e) {
    return { ok: false, status: 500, message: e instanceof Error ? e.message : "汇总加载失败" };
  }

  return {
    ok: true,
    value: {
      items: pageItems.map(({ item, score, daysSince, avgPlay }) => {
        const visibleClaims = Array.isArray(item.sub_topic_claims)
          ? filterTopicClaimsByScope(item.sub_topic_claims as Array<{ user_id?: string | null; status?: string }>, scope)
          : [];
        return {
          ...item,
          sub_topic_claims: visibleClaims,
          summary: summaries.get(String(item.id)) ?? calculateTopicWorkSummary([]),
          claimCount: visibleClaims.filter((c) => c.status !== "returned").length,
          // 额外字段供前端展示
          _score: score,
          _daysSinceLastWork: daysSince,
          _avgPlayCount: avgPlay,
        };
      }),
      pagination: { page: options.page, pageSize: options.pageSize, totalItems },
    },
  };
}
```

### 2.4 新增 `loadNeverWorkedTopics()` 处理 never_worked

```typescript
async function loadNeverWorkedTopics(
  supabase: TopicSupabase,
  userId: string,
  scope: DataAccessScope,
  options: TopicPoolQueryOptions,
): Promise<ApiResult<unknown>> {
  // 取所有有关联 active 作品的子题 id（scope 过滤）
  let worksQuery = supabase
    .from("videos")
    .select("topic_id, user_id")
    .eq("lifecycle_state", "active")
    .not("topic_id", "is", null);
  if (scope.kind !== "all") worksQuery = worksQuery.in("user_id", scope.visibleUserIds);
  const { data: works, error: worksError } = await worksQuery;
  if (worksError) return { ok: false, status: 500, message: worksError.message };

  const workedIds = new Set(
    ((works ?? []) as Array<{ topic_id?: string | null }>)
      .map((w) => w.topic_id)
      .filter((id): id is string => Boolean(id)),
  );

  // 分页查没有作品的子题，按创建时间倒序
  const from = (options.page - 1) * options.pageSize;
  const to = from + options.pageSize - 1;

  let query = supabase
    .from("sub_topics")
    .select("*, topics(id, name, sort_order), topic_groups(id, name, sort_order), sub_topic_claims(id, user_id, status, claimed_at)", { count: "exact" })
    .order("created_at", { ascending: false });

  if (workedIds.size > 0) {
    query = query.not("id", "in", `(${[...workedIds].join(",")})`);
  }

  if (options.topicIds.length > 0) query = query.in("topic_id", options.topicIds);
  if (options.view === "never_worked") {
    // 不做额外过滤，前面已经排除有作品的
  }

  const { data, error, count } = await query.range(from, to);
  if (error) return { ok: false, status: 500, message: error.message };

  const items = (data ?? []) as Array<Record<string, unknown>>;
  return {
    ok: true,
    value: {
      items: items.map((item) => {
        const visibleClaims = Array.isArray(item.sub_topic_claims)
          ? filterTopicClaimsByScope(item.sub_topic_claims as Array<{ user_id?: string | null; status?: string }>, scope)
          : [];
        return {
          ...item,
          sub_topic_claims: visibleClaims,
          summary: calculateTopicWorkSummary([]),
          claimCount: visibleClaims.filter((c) => c.status !== "returned").length,
          _daysSinceLastWork: null,
          _avgPlayCount: null,
        };
      }),
      pagination: { page: options.page, pageSize: options.pageSize, totalItems: count ?? items.length },
    },
  };
}
```

## 三、响应字段说明

新增三个带下划线前缀的扩展字段（前端用于展示，不进数据库）：

| 字段 | 类型 | 说明 |
|---|---|---|
| `_score` | `number` | 综合评分 0~1.0，仅 trending / high_potential 有 |
| `_daysSinceLastWork` | `number \| null` | 最近一次关联作品距今天数，never_worked 为 null |
| `_avgPlayCount` | `number \| null` | 合格作品平均播放量，never_worked 为 null |

## 四、验收要点

1. `GET /api/topics/pool?view=trending` 只返回 30 天内有作品的子题，按综合分倒序，3 天内发布的排在同等播放量的 10 天前发布的前面。
2. `GET /api/topics/pool?view=high_potential` 只返回 30 天以上没发布的子题，60 天以上的综合分高于 30~45 天的（同等播放量时）。
3. `GET /api/topics/pool?view=never_worked` 只返回没有任何关联作品的子题，按创建时间倒序。
4. 三个新 view 均通过 `DataAccessScope` 过滤，普通成员看不到 scope 外的数据。
5. 原有 `all / my_claims / my_created` 三个 view 行为不变。
