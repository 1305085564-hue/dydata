# 数据上传架构根治方案 - 执行规格

> 创建日期：2026-10-13  
> 目标：一次性根治 `report_date` 写入错误及相关架构问题  
> 预计工作量：4.5小时  
> 执行者：Sonnet（新对话）

---

## 背景与痛点

### 痛点1：数据写错了无人知道
- **现状**：`report_date` 可能写错（今天上传3月数据，归属日期变成今天），数据库不拒绝，代码也不检查
- **后果**：数据在历史手稿、月度统计等所有地方都找不到，用户投诉才发现

### 痛点2：改一处漏一处
- **现状**：计算 `report_date` 的代码分散在两个文件：
  - `src/app/api/video-submit/persist.ts` — 新建记录时
  - `src/app/api/video-submit/edit-detail.ts` — 编辑记录时
- **后果**：这次修了第一个文件，忘了第二个文件（后来补上了），但说明架构有问题

### 痛点3：三表写入，失败了数据乱
- **现状**：`videos` → `video_metrics_snapshots` → `daily_reports` 顺序写入，中途失败就停了
- **后果**：`videos` 存在但 `daily_reports` 缺失，数据不完整

---

## 根治目标

### 目标1：数据库强制约束
**即使代码写错，数据库也拒绝**

### 目标2：统一日期计算逻辑
**只有一处代码计算 `report_date`，改一次全生效**

### 目标3：类型系统保护
**写代码时 TypeScript 就会报错，不可能手动传错的 `report_date`**

### 目标4：数据库事务
**三表写入要么全成功，要么全失败**

### 目标5：写入前强制校验
**错误数据提前拦截，不会进数据库**

```typescript
/**
 * 计算日报归属日期
 * 
 * 规则：优先使用视频发布日期（published_at）的上海时区日期，
 *       如果 published_at 为空，则回退到用户输入的业务日期（bizDate）
 * 
 * 数据口径依据：docs/业务逻辑/数据口径与计算规则.md §6
 */
export function computeReportDate(
  publishedAt: string | null | undefined,
  bizDateFallback: string
): string {
  if (!publishedAt) {
    return bizDateFallback;
  }
  
  // 将 UTC 时间转换为上海时区的日期（YYYY-MM-DD）
  const shanghaiDate = new Date(publishedAt)
    .toLocaleString('en-CA', { timeZone: 'Asia/Shanghai' })
    .split(' ')[0];
  
  return shanghaiDate;
}

/**
 * 验证 report_date 是否与 published_at 日期一致
 * 用于写入前校验
 */
export function validateReportDate(
  reportDate: string,
  publishedAt: string | null
): { valid: boolean; error?: string } {
  if (!publishedAt) {
    // published_at 为空时，report_date 可以是任何合法日期
    return { valid: true };
  }
  
  const expectedDate = computeReportDate(publishedAt, '');
  if (reportDate !== expectedDate) {
    return {
      valid: false,
      error: `report_date (${reportDate}) 与 published_at 日期不一致，期望 ${expectedDate}`
    };
  }
  
  return { valid: true };
}
```

### 步骤3：写入前强制校验（1小时）

#### 3.1 在 persist.ts 添加校验

**文件**：`src/app/api/video-submit/persist.ts`

**找到**：`buildDailyReportPayload` 函数

**在函数开头添加校验**：

```typescript
import { computeReportDate, validateReportDate } from "@/lib/video-submit/report-date";

export function buildDailyReportPayload({
  normalized,
  videoId,
  userId,
  submitter,
  nowIso,
  assigneeColumns,
  existingVideo,
}: {
  normalized: NormalizedSubmission;
  videoId: string;
  userId: string;
  submitter: string;
  nowIso: string;
  assigneeColumns: AssigneeColumns;
  existingVideo?: PersistedVideoFact | null;
}) {
  // 计算正确的 report_date
  const reportDate = computeReportDate(
    normalized.mode === "edit" && existingVideo
      ? existingVideo.published_at ?? null
      : normalized.published_at,
    normalized.biz_date
  );

  // 校验（防御性编程：即使类型系统保护了，运行时也再检查一遍）
  const publishedAt = normalized.mode === "edit" && existingVideo
    ? existingVideo.published_at ?? null
    : normalized.published_at;
  
  const validation = validateReportDate(reportDate, publishedAt);
  if (!validation.valid) {
    throw new Error(`数据校验失败：${validation.error}`);
  }

  return {
    user_id: userId,
    report_date: reportDate,
    video_id: videoId,
    title: normalized.video_title || "视频提交",
    // ... 其他字段保持不变
  };
}
```

### 步骤5：数据库事务（2小时）

#### 5.1 创建事务存储过程

**创建 migration**：`supabase/migrations/20261013140000_add_video_submission_transaction.sql`

```sql
-- 创建事务存储过程：原子性写入 videos + snapshots + daily_reports

CREATE OR REPLACE FUNCTION submit_video_with_transaction(
  p_video jsonb,
  p_snapshot jsonb,
  p_report jsonb,
  p_tags jsonb DEFAULT '[]'::jsonb
) RETURNS jsonb AS $$
DECLARE
  v_video_id uuid;
  v_snapshot_id uuid;
  v_report_id uuid;
BEGIN
  -- 第1步：写入 videos
  INSERT INTO videos (
    id, user_id, account_id, video_url, video_title, content,
    published_at, uploaded_at, anomaly_status, punish_type,
    platform_notice, appeal, topic_id, lifecycle_state
  ) VALUES (
    (p_video->>'id')::uuid,
    (p_video->>'user_id')::uuid,
    (p_video->>'account_id')::uuid,
    p_video->>'video_url',
    p_video->>'video_title',
    p_video->>'content',
    (p_video->>'published_at')::timestamptz,
    (p_video->>'uploaded_at')::timestamptz,
    p_video->>'anomaly_status',
    p_video->>'punish_type',
    p_video->>'platform_notice',
    p_video->>'appeal',
    (p_video->>'topic_id')::uuid,
    COALESCE(p_video->>'lifecycle_state', 'active')
  )
  RETURNING id INTO v_video_id;

  -- 第2步：写入 video_metrics_snapshots
  INSERT INTO video_metrics_snapshots (
    video_id, snapshot_type, user_id, account_id,
    play_count, likes, comments, shares, favorites,
    follower_gain, follower_convert,
    completion_rate, avg_play_duration,
    bounce_rate_2s, completion_rate_5s,
    created_at, updated_at
  ) VALUES (
    v_video_id,
    p_snapshot->>'snapshot_type',
    (p_snapshot->>'user_id')::uuid,
    (p_snapshot->>'account_id')::uuid,
    (p_snapshot->>'play_count')::int,
    (p_snapshot->>'likes')::int,
    (p_snapshot->>'comments')::int,
    (p_snapshot->>'shares')::int,
    (p_snapshot->>'favorites')::int,
    (p_snapshot->>'follower_gain')::int,
    (p_snapshot->>'follower_convert')::int,
    (p_snapshot->>'completion_rate')::numeric,
    (p_snapshot->>'avg_play_duration')::numeric,
    (p_snapshot->>'bounce_rate_2s')::numeric,
    (p_snapshot->>'completion_rate_5s')::numeric,
    NOW(),
    NOW()
  )
  RETURNING id INTO v_snapshot_id;

  -- 第3步：写入 daily_reports
  INSERT INTO daily_reports (
    user_id, account_id, video_id, report_date,
    title, content, published_at, uploaded_at,
    play_count, likes, comments, shares, favorites,
    follower_gain, follower_convert,
    completion_rate, avg_play_duration,
    bounce_rate_2s, completion_rate_5s,
    submitter, data_source,
    script_author_user_id, video_editor_user_id, operator_user_id,
    is_void
  ) VALUES (
    (p_report->>'user_id')::uuid,
    (p_report->>'account_id')::uuid,
    v_video_id,
    (p_report->>'report_date')::date,
    p_report->>'title',
    p_report->>'content',
    (p_report->>'published_at')::timestamptz,
    (p_report->>'uploaded_at')::timestamptz,
    (p_report->>'play_count')::int,
    (p_report->>'likes')::int,
    (p_report->>'comments')::int,
    (p_report->>'shares')::int,
    (p_report->>'favorites')::int,
    (p_report->>'follower_gain')::int,
    (p_report->>'follower_convert')::int,
    (p_report->>'completion_rate')::numeric,
    (p_report->>'avg_play_duration')::numeric,
    (p_report->>'bounce_rate_2s')::numeric,
    (p_report->>'completion_rate_5s')::numeric,
    p_report->>'submitter',
    p_report->>'data_source',
    (p_report->>'script_author_user_id')::uuid,
    (p_report->>'video_editor_user_id')::uuid,
    (p_report->>'operator_user_id')::uuid,
    COALESCE((p_report->>'is_void')::boolean, false)
  )
  RETURNING id INTO v_report_id;

  -- 第4步：写入标签（如果有）
  IF jsonb_array_length(p_tags) > 0 THEN
    INSERT INTO video_tags (video_id, tag_dimension, tag_value, confidence, reason, source)
    SELECT
      v_video_id,
      tag->>'tag_dimension',
      tag->>'tag_value',
      (tag->>'confidence')::numeric,
      tag->>'reason',
      COALESCE(tag->>'source', 'ai')
    FROM jsonb_array_elements(p_tags) AS tag;
  END IF;

  -- 返回成功结果
  RETURN jsonb_build_object(
    'success', true,
    'video_id', v_video_id,
    'snapshot_id', v_snapshot_id,
    'report_id', v_report_id
  );

EXCEPTION WHEN OTHERS THEN
  -- 任何失败都回滚，并返回详细错误
  RETURN jsonb_build_object(
    'success', false,
    'error', SQLERRM,
    'detail', SQLSTATE
  );
END;
$$ LANGUAGE plpgsql;

-- 添加注释
COMMENT ON FUNCTION submit_video_with_transaction IS 
'原子性提交视频数据：videos + snapshots + daily_reports + tags，任何一步失败则全部回滚';
```

## 注意事项

### 1. 时区处理
- **重要**：所有日期计算必须使用上海时区（Asia/Shanghai）
- `published_at` 存储的是 UTC 时间，需要转换为上海时区再取日期

### 2. 历史兼容
- `published_at` 为 NULL 的历史记录允许使用 `biz_date`
- 数据库约束中已处理这种情况

### 3. 性能影响
- 事务存储过程可能比原来的顺序写入稍慢（多了一次网络往返）
- 但换来了数据一致性保证，值得

### 4. 错误处理
- 存储过程失败时返回详细错误信息（`SQLERRM` 和 `SQLSTATE`）
- 前端代码要解析这些错误，给用户清晰的提示
