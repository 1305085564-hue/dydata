### 步骤 2：统一身份解析逻辑（1小时）

**新增函数**（在 current-permission-context.ts）：
```typescript
/**
 * 统一的身份解析逻辑
 * getUserPermissions() 和 requireAdminActor() 都调用这个
 */
export async function resolvePermissionCore(): Promise<{
  userId: string;
  companyRole: CompanyRole;
  membershipStatus: "active" | "archived";
  teamId: string | null;
  groupMode: boolean;
  permissions: Permissions;
  dataScope: DataScope;
  visibleUserIds: readonly string[];
  activeVisibleUserIds: readonly string[];
} | null> {
  const { user, authError } = await getCurrentUserContext();
  if (authError || !user) return null;

  const adminSupabase = createAdminClient();
  
  const [profileResult, groupModeState] = await Promise.all([
    adminSupabase
      .from("profiles")
      .select("id, company_role, membership_status, team_id")
      .eq("id", user.id)
      .single(),
    resolveGroupModeForUser(user.id, adminSupabase),
  ]);

  if (profileResult.error || !profileResult.data) return null;
  
  const profile = profileResult.data;
  const companyRole = resolveCompanyRole(profile.company_role);
  if (!companyRole) return null;
  if (profile.membership_status === "archived") return null;
  
  const groupMode = groupModeState.active;
  
  // ✅ 从权限契约获取权限
  const permissionArray = groupMode
    ? PERMISSION_CONTRACT.groupMode.permissions
    : PERMISSION_CONTRACT.roles[companyRole];
  
  const permissions = Object.fromEntries(permissionArray.map(k => [k, true]));
  
  // 数据范围
  const scope = await buildDataAccessScope(adminSupabase, user.id, {
    teamId: profile.team_id,
    profile: {
      id: user.id,
      role: companyRole,
      permissions,
      data_scope: null,
      team_id: profile.team_id,
      company_role: companyRole,
      group_mode: groupMode,
      membership_status: profile.membership_status,
    },
  });
  
  return {
    userId: user.id,
    companyRole,
    membershipStatus: "active",
    teamId: profile.team_id,
    groupMode,
    permissions,
    dataScope: scope.dataScope,
    visibleUserIds: scope.visibleUserIds,
    activeVisibleUserIds: scope.activeVisibleUserIds,
  };
}
```

**改造 requireAdminActor()**：
```typescript
export async function requireAdminActor(
  options: RequireAdminActorOptions = {}
): Promise<RequireAdminActorResult> {
  const core = await resolvePermissionCore();
  
  if (!core) {
    return {
      actor: null,
      error: "未登录或账号已归档",
      statusCode: 401,
    };
  }
  
  if (options.requiredPermission) {
    if (!core.permissions[options.requiredPermission]) {
      return {
        actor: null,
        error: "权限不足",
        statusCode: 403,
      };
    }
  }
  
  return {
    actor: {
      userId: core.userId,
      role: core.companyRole,
      dataScope: core.dataScope,
      teamId: core.teamId,
      groupMode: core.groupMode,
      permissions: core.permissions,
    },
    error: null,
    statusCode: 200,
  };
}
```

```typescript
import type { PermissionKey } from "@/types";

/**
 * 路由权限配置表（单一数据源）
 * 所有 /admin/* 路由必须在此注册
 */
export const ROUTE_PERMISSIONS: Record<string, PermissionKey[]> = {
  "/admin/settings": ["manage_system"],
  "/admin/modules": ["manage_members"],
  "/admin/content": ["review_content"],
  "/admin/videos": ["manage_videos"],
  "/admin/fulfillment": ["manage_fulfillment"],
  "/admin/collaboration": ["view_conversion"],
  "/admin/ai-config": ["use_ai_assist"],
  "/admin": [], // 后台首页：任意 admin 权限即可
} as const;

/**
 * 检查用户是否有权限访问指定路由
 */
export function canAccessRoute(
  pathname: string,
  permissions: Permissions
): boolean {
  const required = ROUTE_PERMISSIONS[pathname];
  
  if (required === undefined) {
    console.warn(`Route ${pathname} not registered in ROUTE_PERMISSIONS`);
    return false;
  }
  
  if (required.length === 0) {
    // 任意权限即可（至少有一个权限为 true）
    return Object.values(permissions).some(v => v === true);
  }
  
  // 检查是否有任意一个权限
  return required.some(perm => permissions[perm] === true);
}
```

```sql
-- 统一的权限获取函数（从 TypeScript permission-contract 同步）
CREATE OR REPLACE FUNCTION get_role_permissions(p_role text, p_group_mode boolean)
RETURNS text[]
LANGUAGE plpgsql
IMMUTABLE
AS $$
BEGIN
  -- 分组模式：全部权限
  IF p_group_mode THEN
    RETURN ARRAY[
      'view_analytics', 'export_data', 'view_conversion',
      'review_content', 'manage_fulfillment', 'manage_videos',
      'manage_members', 'review_violations', 'manage_system',
      'use_ai_copy', 'use_ai_assist'
    ];
  END IF;
  
  -- 按角色返回固定权限
  CASE p_role
    WHEN 'company_owner' THEN
      RETURN ARRAY[
        'view_analytics', 'export_data', 'view_conversion',
        'review_content', 'manage_fulfillment', 'manage_videos',
        'manage_members', 'review_violations', 'manage_system',
        'use_ai_copy', 'use_ai_assist'
      ];
    WHEN 'admin' THEN
      RETURN ARRAY[
        'view_analytics', 'export_data', 'view_conversion',
        'review_content', 'manage_fulfillment', 'manage_videos',
        'manage_members', 'review_violations', 'use_ai_copy'
      ];
    WHEN 'member' THEN
      RETURN ARRAY['view_analytics', 'export_data'];
    ELSE
      RETURN ARRAY[]::text[];
  END CASE;
END;
$$;

-- 重写 has_permission()，使用统一函数
CREATE OR REPLACE FUNCTION has_permission(perm text)
RETURNS boolean
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
AS $$
DECLARE
  v_company_role text;
  v_group_mode_active boolean;
  v_permissions text[];
BEGIN
  -- 获取当前用户的 company_role
  SELECT company_role INTO v_company_role
  FROM profiles
  WHERE id = auth.uid()
    AND coalesce(membership_status, 'active') = 'active';
  
  -- 检查是否在分组模式
  v_group_mode_active := (
    SELECT EXISTS (
      SELECT 1
      FROM group_mode_sessions
      WHERE user_id = auth.uid()
        AND expires_at > now()
    )
  );
  
  -- 调用统一的权限获取函数
  v_permissions := get_role_permissions(v_company_role, v_group_mode_active);
  
  RETURN perm = ANY(v_permissions);
END;
$$;

COMMENT ON FUNCTION get_role_permissions IS '从 TypeScript permission-contract 同步的唯一权限定义';
COMMENT ON FUNCTION has_permission IS '检查当前用户权限（已统一为调用 get_role_permissions）';
```
